df1 <- ga_data(
  propertyId = "357782055",
  date_range = c("2018-01-01", format(Sys.Date(), "%Y-%m-%d")),
  metrics = c("totalUsers"),
  dimensions = c(
    "dayOfWeekName",
    "date",
    "hour",
    "country"
  ),
    limit = -1
)
users_total <- df1 %>%
  summarise(user_count=sum(totalUsers))


map <- df1 %>%
  group_by(country)%>%
  summarise(user_count=sum(totalUsers))%>%
  mutate(iso2c = countrycode(country, origin = 'country.name', destination = 'iso2c'))
